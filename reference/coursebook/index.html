    def export(
        self,
        text: str,
        path: pathlib.Path,
        options: ExportOptions | None = None) -> pathlib.Path:
    """Exports markdown `text` to the file at `path`.

    Args:
        text: pandoc markdown.
        path: path for the exported file.
        options: export options. Defaults to `None`, which uses the
            defaults of `ExportOptions`.

    Returns:
        The path to the exported file.

    """
    options = options or ExportOptions()
    text = _add_metadata(text, self.get_metadata(options))
    pypandoc.convert_text(
        text,
        to = self.pandoc_format,
        format = 'markdown',
        outputfile = str(path),
        extra_args = self.get_arguments(options))
    return path

    def get_arguments(self, options: ExportOptions) -> list[str]:
    """Returns the pandoc command line arguments for `options`.

    Args:
        options: export options.

    Returns:
        The arguments to pass to pandoc.

    """
    arguments: list[str] = []
    if options.template is not None:
        arguments.append(f'--reference-doc={options.template}')
    if options.resource_paths:
        folders = _list_paths(options.resource_paths)
        arguments.append(f'--resource-path={folders}')
    if options.bibliography:
        arguments.append('--citeproc')
        arguments.extend(
            f'--bibliography={path}' for path in options.bibliography)
        if options.csl is not None:
            arguments.append(f'--csl={options.csl}')
    return arguments

    def get_metadata(self, options: ExportOptions) -> dict[str, Any]:
    """Returns the metadata, with the formatting for pandoc's template.

    Args:
        options: export options.

    Returns:
        The metadata in `options` and the template variables and styles
            for `options.formatting`.

    """
    metadata = super().get_metadata(options)
    formatting = options.formatting
    if formatting is None:
        return metadata
    if formatting.body_font:
        # Pandoc reads metadata as markdown and would curl quotation marks,
        # so the font name is not quoted (which CSS allows).
        metadata['mainfont'] = f'{formatting.body_font}, serif'
    if formatting.font_size:
        metadata['fontsize'] = f'{formatting.font_size}pt'
    if formatting.line_spacing:
        metadata['linestretch'] = formatting.line_spacing
    rules = []
    if formatting.heading_font:
        rules.append(
            f'h1, h2, h3, h4, h5, h6 {{ font-family: '
            f'"{formatting.heading_font}", serif; }}')
    if formatting.heading_color:
        rules.append(
            f'h1, h2, h3, h4, h5, h6, .title {{ color: '
            f'#{formatting.heading_color}; }}')
    if formatting.first_line_indent:
        rules.append(
            f'p + p {{ text-indent: {formatting.first_line_indent}in; }}')
    if formatting.center_top_headings:
        rules.append('h1 { text-align: center; }')
    if formatting.capitalize_top_headings:
        rules.append('h1 { text-transform: uppercase; }')
    if rules:
        styles = '\n'.join(rules)
        metadata['header-includes'] = (
            f'```{{=html}}\n<style>\n{styles}\n</style>\n```\n')
    return metadata

    @classmethod
def from_lines(cls, lines: Iterable[OutlineLine]) -> Outline:
    """Creates an outline from labeled lines.

    Args:
        lines: each item is either a line from an outline (such as
            'A: History') or a `tuple` of a label, a title, and
            (optionally) a file. Blank lines are skipped.

    Returns:
        An outline with the level of each entry set from its label.

    Raises:
        ValueError: if a line has no label or title, or if a label cannot
            be placed in the outline.

    """
    outline = cls()
    # The last item is the most recent entry.
    stack: _Stack = []
    for line in lines:
        if isinstance(line, str):
            if not line.strip():
                continue
            label, title, file = parse_outline_line(line)
        else:
            label, title, file = _parse_parts(line)
        if _is_unnumbered(label):
            depth, kind, number = _place_unnumbered(label, title, stack)
            label = ''
        else:
            depth, kind, number = _place_label(label, stack)
        entry = OutlineEntry(
            label = label, title = title, level = depth, file = file)
        del stack[depth:]
        if depth == 0:
            outline.entries.append(entry)
        else:
            stack[-1][0].children.append(entry)
        stack.append((entry, kind, number))
    return outline

    def export(
        self,
        text: str,
        path: pathlib.Path,
        options: ExportOptions | None = None) -> pathlib.Path:
    """Exports markdown `text` to the PDF file at `path`.

    Args:
        text: pandoc markdown.
        path: path for the exported file.
        options: export options. Defaults to `None`, which uses the
            defaults of `ExportOptions`.

    Returns:
        The path to the exported file.

    Raises:
        ImportError: if the `typst` package is not installed.

    """
    try:
        import typst  # noqa: PLC0415
    except ImportError as error:
        message = (
            'PDF export needs the "typst" package (install it with '
            'pip install "coursebook[pdf]")')
        raise ImportError(message) from error
    options = options or ExportOptions()
    with tempfile.TemporaryDirectory() as temporary:
        folder = pathlib.Path(temporary)
        source = folder / 'document.typ'
        # Pandoc runs in the temporary folder so that the images it copies
        # there have paths that Typst can find.
        pypandoc.convert_text(
            _add_metadata(text, self.get_metadata(options)),
            to = self.pandoc_format,
            format = 'markdown',
            outputfile = str(source),
            extra_args = self.get_arguments(options),
            cworkdir = temporary)
        typst.compile(str(source), output = str(path), root = temporary)
    return path

    def get_metadata(self, options: ExportOptions) -> dict[str, Any]:
    """Returns the metadata, with the formatting for pandoc's template.

    Args:
        options: export options.

    Returns:
        The metadata in `options` and the template variables and Typst
            rules for `options.formatting`.

    """
    metadata = super().get_metadata(options)
    metadata['papersize'] = self.paper_size
    formatting = options.formatting
    if formatting is None:
        return metadata
    if formatting.body_font:
        metadata['mainfont'] = formatting.body_font
    if formatting.font_size:
        metadata['fontsize'] = f'{formatting.font_size}pt'
    if formatting.line_spacing:
        metadata['linestretch'] = formatting.line_spacing
    if formatting.margins is not None:
        margin = f'{formatting.margins}in'
        metadata['margin'] = {'x': margin, 'y': margin}
    if formatting.page_numbers:
        metadata['page-numbering'] = '1'
    rules = []
    if formatting.heading_font:
        rules.append(
            f'#show heading: set text(font: "{formatting.heading_font}")')
    if formatting.heading_color:
        rules.append(
            f'#show heading: set text(fill: '
            f'rgb("#{formatting.heading_color}"))')
    if formatting.first_line_indent:
        rules.append(
            f'#set par(first-line-indent: '
            f'{formatting.first_line_indent}in)')
    if formatting.footnote_size:
        rules.append(
            f'#show footnote.entry: set text(size: '
            f'{formatting.footnote_size}pt)')
    if formatting.center_top_headings:
        rules.append('#show heading.where(level: 1): set align(center)')
    if formatting.capitalize_top_headings:
        rules.append('#show heading.where(level: 1): upper')
    if rules:
        typst_rules = '\n'.join(rules)
        metadata['header-includes'] = (
            f'```{{=typst}}\n{typst_rules}\n```\n')
    return metadata

    def build(
        self,
        include: Iterable[str] | None = None,
        output_folder: _PathLike | None = None) -> list[pathlib.Path]:
    """Creates the files for the sections of the project.

    Each section is exported to every format set for it (such as Word and
    PDF). The files are exported at the same time, each in its own pandoc
    process.

    Args:
        include: the sections to create files for. Defaults to `None`,
            which uses `include`.
        output_folder: folder for the created files. Defaults to `None`,
            which uses `output_folder`.

    Returns:
        The paths to the created files, in the order of `include` and then
            of the formats of each section.

    Raises:
        FileNotFoundError: if a template, bibliography, or CSL file is not
            found.
        ImportError: if a format needs a package that is not installed.
        ValueError: if a section, format, or template is not valid, or a
            format cannot hold a section.
        RuntimeError: if pandoc fails.

    """
    sections = settings.normalize_sections(
        self.include if include is None else include)
    folder = (
        self.output_folder if output_folder is None
        else self._resolve(output_folder))
    for entry in self.missing:
        _LOGGER.warning(
            'No source file "%s" for outline entry "%s"',
            entry.file_name, entry.title)
    folder.mkdir(parents = True, exist_ok = True)
    stem = markdown.normalize_name(self.settings.name) or 'coursebook'
    with tempfile.TemporaryDirectory() as temporary:
        jobs = []
        for section in sections:
            text = self.assemble(section)
            for name in self.settings.formats_for(section):
                exporter = exporters.get_exporter(name)
                if section not in exporter.sections:
                    message = (
                        f'the {section} cannot be exported to {name} '
                        f'(use {", ".join(_formats_for(section))})')
                    raise ValueError(message)
                options = self._get_options(
                    section, exporter, pathlib.Path(temporary))
                path = folder / (
                    f'{stem}{_SUFFIXES[section]}.{exporter.extension}')
                jobs.append((exporter, text, path, options))
        with concurrent.futures.ThreadPoolExecutor() as pool:
            futures = [pool.submit(job[0].export, *job[1:]) for job in jobs]
            return [future.result() for future in futures]

    def get_exporter(name: str) -> Exporter:
    """Returns the exporter with `name`.

    Args:
        name: the name of an `Exporter` subclass or one of its aliases (such as
            'word' or 'PowerPoint').

    Returns:
        An instance of the exporter.

    Raises:
        ValueError: if no exporter has `name`.

    """
    key = markdown.normalize_name(name)
    choices: dict[str, type[Exporter]] = {}
    for registered_name, registered in Exporter.registry.items():
        choices[str(registered_name)] = registered
        for alias in registered.aliases:
            choices.setdefault(alias, registered)
    choices.pop('exporter', None)
    if key not in choices:
        known = ', '.join(sorted(
            alias for exporter in choices.values()
            for alias in exporter.aliases))
        message = f'unknown format "{name}" (use one of: {known})'
        raise ValueError(message)
    return choices[key]()

    def get_style(name: str, **overrides: Any) -> Style:
    """Returns the style with `name`.

    Args:
        name: the name of a `Style` subclass or one of its aliases (such as
            'law review' or 'Coursebook').
        **overrides: values that replace the defaults of the style. Values
            of `None` are ignored.

    Returns:
        An instance of the style.

    Raises:
        ValueError: if no style has `name`.

    """
    key = markdown.normalize_name(name)
    choices: dict[str, type[Style]] = {}
    for registered_name, registered in Style.registry.items():
        choices[str(registered_name)] = registered
        for alias in registered.aliases:
            choices.setdefault(alias, registered)
    choices.pop('style', None)
    if key not in choices:
        known = ', '.join(sorted(choices))
        message = f'unknown style "{name}" (use one of: {known})'
        raise ValueError(message)
    kwargs = {
        field: value for field, value in overrides.items()
        if value is not None}
    return choices[key](**kwargs)
